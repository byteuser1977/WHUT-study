```c

int main(void){
    int n, i, x, a[11];
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    scanf("%d", &x);
    for (i = n; i > 0 && a[i - 1] > x; i--)

    return 0;
}
```
