## 1. 宏定义

宏是**文本替换**，在编译前由预处理器处理。

### 1.1 不带参数的宏

```c
// 格式：#define 宏名 替换文本
#define PI 3.14159
#define MAX_SIZE 100
#define GREETING "Hello, World!"

// 使用
double area = PI * 5 * 5;
int arr[MAX_SIZE];
printf("%s\n", GREETING);
```

### 1.2 带参数的宏

```c
// 格式：#define 宏名(参数) 替换文本
#define SQUARE(x) ((x) * (x))
#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define MIN(a, b) ((a) < (b) ? (a) : (b))

// 使用
int area = SQUARE(5);  // 展开为 ((5) * (5))
int bigger = MAX(10, 20);  // 展开为 ((10) > (20) ? (10) : (20))
```

### 1.3 宏的陷阱

```c
// 陷阱1：缺少括号
#define SQUARE_BAD(x) x * x
int result = SQUARE_BAD(3 + 2);  // 展开为 3 + 2 * 3 + 2 = 11，不是25！

// 正确写法
#define SQUARE_GOOD(x) ((x) * (x))
int result = SQUARE_GOOD(3 + 2);  // 展开为 ((3 + 2) * (3 + 2)) = 25

// 陷阱2：副作用参数
#define SQUARE(x) ((x) * (x))
int a = 5;
int result = SQUARE(a++);  // 展开为 ((a++) * (a++))，a被自增两次！
```

### 1.4 多行宏

```c
// 使用反斜杠 \ 换行
#define PRINT_SUM(a, b) \
    do { \
        int sum = (a) + (b); \
        printf("%d + %d = %d\n", (a), (b), sum); \
    } while(0)

// 使用 do-while(0) 包装，确保宏在任何上下文中都像一条语句
```

## 2. 文件包含

### 2.1 两种包含方式

```c
// 尖括号：在系统头文件目录中查找
#include <stdio.h>

// 双引号：先在当前目录查找，再在系统目录查找
#include "myheader.h"
```

### 2.2 防止重复包含

```c
// 方法1：ifndef 守卫
#ifndef MYHEADER_H
#define MYHEADER_H

// 头文件内容

#endif

// 方法2：pragma once（非标准但广泛支持）
#pragma once

// 头文件内容
```

## 3. 条件编译

根据条件决定是否编译某段代码。

### 3.1 #ifdef / #ifndef

```c
// #ifdef：如果定义了宏
#ifdef DEBUG
    printf("调试信息：%d\n", value);
#endif

// #ifndef：如果没有定义宏
#ifndef RELEASE
    printf("开发版本\n");
#endif

// 常用于防止头文件重复包含
#ifndef MYHEADER_H
#define MYHEADER_H
// 头文件内容
#endif
```

### 3.2 #if / #elif / #else / #endif

```c
#define VERSION 3

#if VERSION == 1
    printf("版本1\n");
#elif VERSION == 2
    printf("版本2\n");
#elif VERSION == 3
    printf("版本3\n");
#else
    printf("未知版本\n");
#endif
```

### 3.3 条件编译的应用

```c
// 调试开关
#define DEBUG 1

#if DEBUG
    #define LOG(msg) printf("[DEBUG] %s\n", msg)
#else
    #define LOG(msg)  // 空操作
#endif

// 使用
LOG("程序开始");  // DEBUG=1时输出，DEBUG=0时无操作

// 跨平台编译
#ifdef _WIN32
    printf("Windows系统\n");
#elif __linux__
    printf("Linux系统\n");
#elif __APPLE__
    printf("macOS系统\n");
#endif
```

## 4. 预定义宏

```c
#include <stdio.h>

int main(void) {
    printf("文件名：%s\n", __FILE__);
    printf("行号：%d\n", __LINE__);
    printf("日期：%s\n", __DATE__);
    printf("时间：%s\n", __TIME__);
    printf("函数名：%s\n", __func__);  // C99
    
    return 0;
}
```

### 必做（动手敲，不许复制粘贴）

1. **安全的交换宏**：
   ```c
   #include <stdio.h>
   #define SWAP(a, b) \
       do { \
           typeof(a) temp = (a); \
           (a) = (b); \
           (b) = temp; \
       } while(0)
   
   int main(void) {
       int x = 10, y = 20;
       printf("交换前：x=%d, y=%d\n", x, y);
       SWAP(x, y);
       printf("交换后：x=%d, y=%d\n", x, y);
       return 0;
   }
   ```

2. **调试日志宏**：
   ```c
   #include <stdio.h>
   #define DEBUG_MODE 1
   
   #if DEBUG_MODE
       #define LOG(fmt, ...) printf("[LOG] " fmt "\n", ##__VA_ARGS__)
   #else
       #define LOG(fmt, ...)
   #endif
   
   int main(void) {
       int value = 42;
       LOG("value = %d", value);
       LOG("程序运行中");
       return 0;
   }
   ```

3. **条件编译练习**：

   #include <stdio.h>

           printf("Linux路径分隔符：/\n");
       #elif PLATFORM == 3
           printf("macOS路径分隔符：/\n");
       #else
           printf("未知平台\n");
       #endif
       return 0;
   }

## 本讲小结

| 概念 | 要点 |
|------|------|
| 宏定义 | 文本替换，编译前处理 |
| 不带参数宏 | `#define 宏名 值` |
| 带参数宏 | `#define 宏名(参数) 表达式`，注意加括号 |
| 文件包含 | `< >` 系统目录，`" "` 当前目录 |
| 防重复包含 | `#ifndef` 守卫或 `#pragma once` |
| 条件编译 | `#ifdef/#ifndef/#if/#elif/#else/#endif` |
| 预定义宏 | `__FILE__`, `__LINE__`, `__DATE__`, `__TIME__`, `__func__` |
