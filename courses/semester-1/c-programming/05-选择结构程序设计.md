## 1. if 语句

程序可以根据条件决定执行哪段代码，就像岔路口选择方向。

```c
// 基本形式
if (条件) {
    // 条件为真（非0）时执行
}

// 示例
int age = 18;
if (age >= 18) {
    printf("你已成年\n");
}
```

---

## 2. if-else 语句

```c
// 二选一
if (条件) {
    // 条件为真时执行
} else {
    // 条件为假时执行
}

// 示例
int score = 85;
if (score >= 60) {
    printf("及格\n");
} else {
    printf("不及格\n");
}
```

---

## 3. 嵌套 if 语句

```c
// 多层判断
if (条件1) {
    if (条件2) {
        // 条件1和条件2都为真
    } else {
        // 条件1为真，条件2为假
    }
} else {
    // 条件1为假
}

// 示例：判断成绩等级
int score = 85;
if (score >= 90) {
    printf("优秀\n");
} else {
    if (score >= 80) {
        printf("良好\n");
    } else {
        if (score >= 70) {
            printf("中等\n");
        } else {
            if (score >= 60) {
                printf("及格\n");
            } else {
                printf("不及格\n");
            }
        }
    }
}
```

### 3.1 else-if 链（简化嵌套）

```c
// 更清晰的写法
int score = 85;
if (score >= 90) {
    printf("优秀\n");
} else if (score >= 80) {
    printf("良好\n");
} else if (score >= 70) {
    printf("中等\n");
} else if (score >= 60) {
    printf("及格\n");
} else {
    printf("不及格\n");
}
```

## 4. switch-case 语句

当需要对**同一个变量**进行多个值判断时，switch比if-else链更清晰。

```c
switch (表达式) {
    case 常量1:
        // 表达式 == 常量1 时执行
        break;
    case 常量2:
        // 表达式 == 常量2 时执行
        break;
    default:
        // 以上都不匹配时执行
        break;
}

// 示例：根据数字输出星期
int day = 3;
switch (day) {
    case 1:
        printf("星期一\n");
        break;
    case 2:
        printf("星期二\n");
        break;
    case 3:
        printf("星期三\n");
        break;
    case 4:
        printf("星期四\n");
        break;
    case 5:
        printf("星期五\n");
        break;
    case 6:
        printf("星期六\n");
        break;
    case 7:
        printf("星期日\n");
        break;
    default:
        printf("无效的输入\n");
        break;
}
```

### 4.1 switch 穿透（fall-through）

```c
// 不写break会"穿透"到下一个case
int num = 2;
switch (num) {
    case 1:
        printf("一\n");
        // 没有break，继续执行
    case 2:
        printf("二\n");
        // 没有break，继续执行
    case 3:
        printf("三\n");
        break;
}
// 输出：二 三

// 利用穿透实现多值匹配
int month = 2;
switch (month) {
    case 1: case 3: case 5: case 7:
    case 8: case 10: case 12:
        printf("31天\n");
        break;
    case 4: case 6: case 9: case 11:
        printf("30天\n");
        break;
    case 2:
        printf("28或29天\n");
        break;
}
```

## 5. 条件运算符（三目运算符）

```c
// 简化的 if-else
// 格式：条件 ? 表达式1 : 表达式2

int a = 10, b = 20;
int max = (a > b) ? a : b;  // max = 20

// 等价于
int max;
if (a > b) {
    max = a;
} else {
    max = b;
}

// 可嵌套（不推荐，可读性差）
int sign = (x > 0) ? 1 : (x < 0) ? -1 : 0;
```

### 必做（动手敲，不许复制粘贴）

1. **判断奇偶数**：
   ```c
   #include <stdio.h>
   int main(void) {
       int num;
       printf("请输入一个整数：");
       scanf("%d", &num);
       if (num % 2 == 0) {
           printf("%d是偶数\n", num);
       } else {
           printf("%d是奇数\n", num);
       }
       return 0;
   }
   ```

2. **计算分段函数**：
   ```c
   #include <stdio.h>
   int main(void) {
       double x, y;
       printf("请输入x的值：");
       scanf("%lf", &x);
       if (x < 0) {
           y = 0;
       } else if (x <= 10) {
           y = x;
       } else {
           y = 10;
       }
       printf("y = %.2f\n", y);
       return 0;
   }
   ```

3. **简单计算器**：
   ```c
   #include <stdio.h>
   int main(void) {
       double num1, num2, result;
       char op;
       printf("请输入表达式（如 3 + 5）：");
       scanf("%lf %c %lf", &num1, &op, &num2);
       switch (op) {
           case '+':
               result = num1 + num2;
               break;
           case '-':
               result = num1 - num2;
               break;
           case '*':
               result = num1 * num2;
               break;
           case '/':
               if (num2 != 0) {
                   result = num1 / num2;
               } else {
                   printf("错误：除数不能为0\n");
                   return 1;
               }
               break;
           default:
               printf("错误：不支持的运算符\n");
               return 1;
       }
       printf("%.2f %c %.2f = %.2f\n", num1, op, num2, result);
       return 0;
   }
   ```

## 本讲小结

| 概念 | 要点 |
|------|------|
| if 语句 | 单分支，条件为真执行 |
| if-else | 双分支，二选一 |
| 嵌套 if | 多层判断，注意 else 匹配 |
| else-if 链 | 多分支简化写法 |
| switch-case | 多值匹配，记得 break |
| switch 穿透 | 不写 break 会继续执行下一个 case |
| 条件运算符 | `条件 ? 表达式1 : 表达式2` |
