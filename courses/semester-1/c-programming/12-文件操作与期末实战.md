## 1. 文件基础

### 1.1 文件指针

```c
#include <stdio.h>

FILE *fp;  // 文件指针
```

### 1.2 打开文件

```c
// fopen 返回文件指针，失败返回 NULL
fp = fopen("data.txt", "r");  // 只读
fp = fopen("data.txt", "w");  // 只写（创建/清空）
fp = fopen("data.txt", "a");  // 追加
fp = fopen("data.txt", "r+"); // 读写

if (fp == NULL) {
    printf("打开文件失败！\n");
    return 1;
}
```

### 1.3 关闭文件

```c
fclose(fp);  // 操作完必须关闭
```

---

## 2. 文件读写

### 2.1 字符读写

```c
// 写入单个字符
fputc('A', fp);

// 读取单个字符
int ch = fgetc(fp);  // 读到文件结束返回 EOF
```

### 2.2 字符串读写

```c
// 写入字符串
fputs("Hello, World!\n", fp);

// 读取字符串
char buffer[100];
fgets(buffer, sizeof(buffer), fp);
```

### 2.3 格式化读写

```c
// 格式化写入
fprintf(fp, "姓名：%s，年龄：%d\n", name, age);

// 格式化读取
char name[50];
int age;
fscanf(fp, "姓名：%s，年龄：%d", name, &age);
```

### 2.4 二进制读写

```c
// 写入结构体数组
fwrite(students, sizeof(Student), count, fp);

// 读取结构体数组
fread(students, sizeof(Student), count, fp);
```

---

## 3. 文件定位

```c
// 获取当前位置
long pos = ftell(fp);

// 移动到文件开头
rewind(fp);

// 移动到指定位置
fseek(fp, 0, SEEK_SET);  // 文件开头
fseek(fp, 10, SEEK_CUR); // 当前位置+10
fseek(fp, -5, SEEK_END); // 文件结尾-5
```

---

## 4. 错误处理

```c
// 检查文件结束
while (!feof(fp)) {
    // 读取操作
}

// 检查错误
if (ferror(fp)) {
    printf("文件操作出错！\n");
}

// 获取错误码
int err = ferror(fp);
```

## 5. 期末实战：简单通讯录

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_CONTACTS 100
#define FILE_NAME "contacts.dat"

// 联系人结构
typedef struct {
    char name[50];
    char phone[20];
    char email[50];
} Contact;

// 通讯录
typedef struct {
    Contact contacts[MAX_CONTACTS];
    int count;
} AddressBook;

// 保存到文件
void save_to_file(const AddressBook *book) {
    FILE *fp = fopen(FILE_NAME, "wb");
    if (fp == NULL) {
        printf("保存失败！\n");
        return;
    }
    fwrite(book, sizeof(AddressBook), 1, fp);
    fclose(fp);
    printf("保存成功！\n");
}

// 从文件加载
void load_from_file(AddressBook *book) {
    FILE *fp = fopen(FILE_NAME, "rb");
    if (fp == NULL) {
        book->count = 0;
        return;
    }
    fread(book, sizeof(AddressBook), 1, fp);
    fclose(fp);
    printf("加载成功，共有 %d 个联系人\n", book->count);
}

// 添加联系人
void add_contact(AddressBook *book) {
    if (book->count >= MAX_CONTACTS) {
        printf("通讯录已满！\n");
        return;
    }
    
    Contact *c = &book->contacts[book->count];
    printf("请输入姓名：");
    scanf("%s", c->name);
    printf("请输入电话：");
    scanf("%s", c->phone);
    printf("请输入邮箱：");
    scanf("%s", c->email);
    
    book->count++;
    printf("添加成功！\n");
}

// 显示所有联系人
void show_contacts(const AddressBook *book) {
    if (book->count == 0) {
        printf("通讯录为空！\n");
        return;
    }
    
    printf("\n===== 通讯录 =====\n");
    for (int i = 0; i < book->count; i++) {
        const Contact *c = &book->contacts[i];
        printf("%d. %s | %s | %s\n", i + 1, c->name, c->phone, c->email);
    }
    printf("==================\n");
}

// 查找联系人
void search_contact(const AddressBook *book) {
    char name[50];
    printf("请输入要查找的姓名：");
    scanf("%s", name);
    
    for (int i = 0; i < book->count; i++) {
        if (strcmp(book->contacts[i].name, name) == 0) {
            printf("找到：%s | %s | %s\n",
                   book->contacts[i].name,
                   book->contacts[i].phone,
                   book->contacts[i].email);
            return;
        }
    }
    printf("未找到该联系人！\n");
}

// 删除联系人
void delete_contact(AddressBook *book) {
    char name[50];
    printf("请输入要删除的姓名：");
    scanf("%s", name);
    
    for (int i = 0; i < book->count; i++) {
        if (strcmp(book->contacts[i].name, name) == 0) {
            // 移动后续元素
            for (int j = i; j < book->count - 1; j++) {
                book->contacts[j] = book->contacts[j + 1];
            }
            book->count--;
            printf("删除成功！\n");
            return;
        }
    }
    printf("未找到该联系人！\n");
}

// 显示菜单
void show_menu(void) {
    printf("\n===== 通讯录管理系统 =====\n");
    printf("1. 添加联系人\n");
    printf("2. 显示所有联系人\n");
    printf("3. 查找联系人\n");
    printf("4. 删除联系人\n");
    printf("5. 保存并退出\n");
    printf("请选择：");
}

int main(void) {
    AddressBook book;
    load_from_file(&book);
    
    int choice;
    do {
        show_menu();
        scanf("%d", &choice);
        
        switch (choice) {
            case 1: add_contact(&book); break;
            case 2: show_contacts(&book); break;
            case 3: search_contact(&book); break;
            case 4: delete_contact(&book); break;
            case 5: save_to_file(&book); break;
            default: printf("无效选择！\n"); break;
        }
    } while (choice != 5);
    
    return 0;
}
```

1. **文件复制程序**：
   ```c
   #include <stdio.h>
   
   int main(int argc, char *argv[]) {
       if (argc != 3) {
           printf("用法：copy 源文件 目标文件\n");
           return 1;
       }
       
       FILE *src = fopen(argv[1], "rb");
       if (src == NULL) {
           printf("无法打开源文件！\n");
           return 1;
       }
       
       FILE *dst = fopen(argv[2], "wb");
       if (dst == NULL) {
           printf("无法创建目标文件！\n");
           fclose(src);
           return 1;
       }
       
       char buffer[1024];
       size_t bytes;
       while ((bytes = fread(buffer, 1, sizeof(buffer), src)) > 0) {
           fwrite(buffer, 1, bytes, dst);
       }
       
       fclose(src);
       fclose(dst);
       printf("复制完成！\n");
       
       return 0;
   }
   ```

2. **统计文件中的字符数、单词数、行数**：
   ```c
   #include <stdio.h>
   #include <ctype.h>
   
   int main(int argc, char *argv[]) {
       if (argc != 2) {
           printf("用法：count 文件名\n");
           return 1;
       }
       
       FILE *fp = fopen(argv[1], "r");
       if (fp == NULL) {
           printf("无法打开文件！\n");
           return 1;
       }
       
       int chars = 0, words = 0, lines = 0;
       int in_word = 0;
       int ch;
       
       while ((ch = fgetc(fp)) != EOF) {
           chars++;
           if (ch == '\n') lines++;
           if (isspace(ch)) {
               in_word = 0;
           } else if (!in_word) {
               in_word = 1;
               words++;
           }
       }
       
       fclose(fp);
       printf("字符数：%d\n", chars);
       printf("单词数：%d\n", words);
       printf("行数：%d\n", lines);
       
       return 0;
   }
   ```
