## 1. 结构体

结构体是**不同类型数据的集合**，就像表格中的一行。

### 1.1 定义和初始化

```c
// 定义结构体类型
struct Student {
    char name[50];
    int age;
    float score;
};

// 声明变量并初始化
struct Student s1 = {"Alice", 20, 95.5};

// 声明变量后赋值
struct Student s2;
strcpy(s2.name, "Bob");  // 字符串不能直接赋值
s2.age = 21;
s2.score = 88.0;

// C99：指定初始化器
struct Student s3 = {
    .name = "Charlie",
    .age = 19,
    .score = 92.0
};
```

### 1.2 访问成员

```c
// 使用 . 运算符
printf("姓名：%s\n", s1.name);
printf("年龄：%d\n", s1.age);
printf("成绩：%.1f\n", s1.score);
```

### 1.3 typedef 简化

```c
// 给结构体类型起别名
typedef struct {
    char name[50];
    int age;
    float score;
} Student;

// 现在可以直接用 Student 声明变量
Student s1 = {"Alice", 20, 95.5};
```

## 2. 结构体数组

```c
// 多个学生信息
Student class[3] = {
    {"Alice", 20, 95.5},
    {"Bob", 21, 88.0},
    {"Charlie", 19, 92.0}
};

// 遍历数组
for (int i = 0; i < 3; i++) {
    printf("%s: %.1f\n", class[i].name, class[i].score);
}

// 计算平均分
float sum = 0;
for (int i = 0; i < 3; i++) {
    sum += class[i].score;
}
printf("平均分：%.1f\n", sum / 3);
```

## 3. 结构体指针

### 3.1 指针访问成员

```c
Student s = {"Alice", 20, 95.5};
Student *p = &s;

// 两种方式访问成员
printf("姓名：%s\n", (*p).name);   // 解引用后用.
printf("姓名：%s\n", p->name);    // 箭头运算符（推荐）
```

### 3.2 结构体传参

```c
// 值传递：复制整个结构体
void print_student(Student s) {
    printf("%s: %.1f\n", s.name, s.score);
}

// 指针传递：只复制地址（高效）
void print_student_ptr(const Student *s) {
    printf("%s: %.1f\n", s->name, s->score);
}

// 修改结构体
void update_score(Student *s, float new_score) {
    s->score = new_score;
}
```

## 4. 结构体嵌套

```c
// 日期结构
typedef struct {
    int year;
    int month;
    int day;
} Date;

// 学生结构（嵌套日期）
typedef struct {
    char name[50];
    Date birthday;
    float score;
} Student;

// 访问嵌套成员
Student s = {
    "Alice",
    {2004, 5, 15},
    95.5
};
printf("生日：%d-%02d-%02d\n", 
       s.birthday.year, s.birthday.month, s.birthday.day);
```

## 5. 共用体（联合体）

共用体的**所有成员共享同一块内存**，大小等于最大成员的大小。

```c
union Data {
    int i;
    float f;
    char c;
};

int main(void) {
    union Data d;
    printf("大小：%zu\n", sizeof(d));  // 4（最大成员int的大小）
    
    d.i = 42;
    printf("d.i = %d\n", d.i);
    
    d.f = 3.14;  // 覆盖了d.i
    printf("d.f = %.2f\n", d.f);
    printf("d.i = %d\n", d.i);  // 已被覆盖，值无意义
    
    return 0;
}
```

### 5.1 共用体的应用

```c
// 类型双关（Type Punning）
union {
    float f;
    unsigned int bits;
} converter;

converter.f = 3.14f;
printf("浮点数的二进制表示：0x%X\n", converter.bits);

// 判断大小端
union {
    unsigned int num;
    unsigned char bytes[4];
} endian_test;

endian_test.num = 0x12345678;
if (endian_test.bytes[0] == 0x78) {
    printf("小端序\n");
} else {
    printf("大端序\n");
}
```

## 6. 枚举

枚举用于定义**命名的整数常量**。

```c
// 定义枚举类型
enum Weekday {
    SUNDAY,    // 0
    MONDAY,    // 1
    TUESDAY,   // 2
    WEDNESDAY, // 3
    THURSDAY,  // 4
    FRIDAY,    // 5
    SATURDAY   // 6
};

// 使用枚举
enum Weekday today = WEDNESDAY;
printf("今天是星期%d\n", today);  // 输出3

// 指定值
enum Color {
    RED = 1,
    GREEN = 3,
    BLUE = 5
};

// 枚举用于状态码
enum Status {
    OK = 0,
    ERROR_FILE_NOT_FOUND = -1,
    ERROR_PERMISSION_DENIED = -2
};
```

## 7. 位域

结构体中可以指定成员占用的位数。

```c
struct Flags {
    unsigned int enable    : 1;  // 1位，值0或1
    unsigned int mode      : 3;  // 3位，值0-7
    unsigned int priority  : 4;  // 4位，值0-15
};

printf("大小：%zu\n", sizeof(struct Flags));  // 4字节
```

   ```c
   #include <stdio.h>
   #include <string.h>
   
   typedef struct {
       char name[50];
       int id;
       float scores[3];  // 三门课成绩
   } Student;
   
   float get_average(const Student *s) {
       return (s->scores[0] + s->scores[1] + s->scores[2]) / 3.0;
   }
   
   void print_student(const Student *s) {
       printf("学号：%d，姓名：%s\n", s->id, s->name);
       printf("成绩：%.1f, %.1f, %.1f，平均：%.1f\n",
              s->scores[0], s->scores[1], s->scores[2],
              get_average(s));
   }

       Student students[] = {
           {"Alice", 1001, {90, 85, 92}},
           {"Bob", 1002, {78, 82, 88}},
           {"Charlie", 1003, {95, 91, 89}}
       };
       int size = sizeof(students) / sizeof(students[0]);

           print_student(&students[i]);
           printf("\n");
       }
       
       return 0;
   }
   ```

2. **枚举实现简单状态机**：
   ```c
   #include <stdio.h>
   
   typedef enum {
       STATE_OFF,
       STATE_ON,
       STATE_ERROR
   } State;
   
   const char* state_to_string(State s) {
       switch (s) {
           case STATE_OFF: return "关闭";
           case STATE_ON: return "开启";
           case STATE_ERROR: return "错误";
           default: return "未知";
       }
   }
   
   int main(void) {
       State current = STATE_OFF;
       printf("当前状态：%s\n", state_to_string(current));
       
       current = STATE_ON;
       printf("当前状态：%s\n", state_to_string(current));
       
       current = STATE_ERROR;
       printf("当前状态：%s\n", state_to_string(current));
       
       return 0;
   }
   ```

3. **共用体判断大小端**：
   ```c
   #include <stdio.h>
   
   int is_little_endian(void) {
       union {
           unsigned int num;
           unsigned char bytes[4];
       } test;
       test.num = 0x12345678;
       return test.bytes[0] == 0x78;
   }
   
   int main(void) {
       if (is_little_endian()) {
           printf("小端序\n");
       } else {
           printf("大端序\n");
       }
       return 0;
   }
   ```

## 本讲小结

| 概念 | 要点 |
|------|------|
| 结构体 | 不同类型数据的集合，用 `.` 访问成员 |
| typedef | 给类型起别名 |
| 结构体指针 | 用 `->` 访问成员 |
| 结构体传参 | 推荐指针传递（高效） |
| 共用体 | 成员共享内存，大小等于最大成员 |
| 枚举 | 命名的整数常量 |
| 位域 | 指定成员占用的位数 |
