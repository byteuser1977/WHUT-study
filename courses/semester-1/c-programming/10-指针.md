## 1. 什么是指针

指针是**存储内存地址的变量**。

```c
// 类比：变量是盒子，指针是"盒子的地址标签"
int a = 10;      // a 是变量，存储值 10
int *p = &a;     // p 是指针，存储 a 的地址

// & 取地址运算符：获取变量的地址
printf("a 的地址：%p\n", (void*)&a);
printf("p 的值：%p\n", (void*)p);  // 两者相同

// * 解引用运算符：通过地址访问值
printf("p 指向的值：%d\n", *p);  // 输出 10

*p = 20;  // 通过指针修改 a 的值
printf("a = %d\n", a);  // 输出 20
```

## 2. 指针与数组

数组名在大多数情况下会退化为指向首元素的指针。

```c
int arr[5] = {10, 20, 30, 40, 50};
int *p = arr;  // arr 退化为 &arr[0]

// 通过指针访问数组
printf("arr[0] = %d\n", *p);        // 10
printf("arr[1] = %d\n", *(p + 1));  // 20
printf("arr[2] = %d\n", p[2]);      // 30（指针也支持下标）

// 指针遍历数组
for (int i = 0; i < 5; i++) {
    printf("%d ", *(p + i));
}
printf("\n");

// 指针算术
p++;  // 指向下一个元素
printf("现在指向：%d\n", *p);  // 20
```

## 3. 指针作为函数参数

### 3.1 交换两个数

```c
// 值传递：无法交换
void swap_wrong(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

// 指针传递：可以交换
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 10, y = 20;
    swap(&x, &y);  // 传递地址
    printf("x=%d, y=%d\n", x, y);  // x=20, y=10
    return 0;
}
```

### 3.2 返回多个值

```c
// C语言函数只能返回一个值，但可以通过指针"返回"多个
void get_min_max(int arr[], int size, int *min, int *max) {
    *min = arr[0];
    *max = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] < *min) *min = arr[i];
        if (arr[i] > *max) *max = arr[i];
    }
}

int main(void) {
    int arr[] = {3, 1, 4, 1, 5, 9, 2, 6};
    int min, max;
    get_min_max(arr, 8, &min, &max);
    printf("最小值：%d，最大值：%d\n", min, max);
    return 0;
}
```

## 4. 指针数组

```c
// 指针数组：数组的每个元素都是指针
int a = 10, b = 20, c = 30;
int *arr[3] = {&a, &b, &c};  // 指针数组

// 字符串数组（二维字符数组的替代）
const char *names[] = {
    "Alice",
    "Bob",
    "Charlie"
};
int len = sizeof(names) / sizeof(names[0]);
for (int i = 0; i < len; i++) {
    printf("names[%d] = %s\n", i, names[i]);
}
```

## 5. 指向函数的指针

```c
// 函数指针：存储函数的地址
int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }

int main(void) {
    // 声明函数指针
    int (*operation)(int, int);
    
    operation = add;  // 指向add函数
    printf("3 + 5 = %d\n", operation(3, 5));
    
    operation = sub;  // 指向sub函数
    printf("10 - 4 = %d\n", operation(10, 4));
    
    return 0;
}

// 函数指针作为参数（回调函数）
void apply(int a, int b, int (*func)(int, int)) {
    printf("结果：%d\n", func(a, b));
}

apply(3, 5, add);   // 结果：8
apply(10, 4, sub);  // 结果：6
```

## 6. void 指针

```c
// void* 可以指向任何类型的指针
void print_value(void *ptr, char type) {
    switch (type) {
        case 'i': printf("值：%d\n", *(int*)ptr); break;
        case 'f': printf("值：%.2f\n", *(float*)ptr); break;
        case 'c': printf("值：%c\n", *(char*)ptr); break;
    }
}

int main(void) {
    int x = 42;
    float y = 3.14;
    char z = 'A';
    
    print_value(&x, 'i');  // 值：42
    print_value(&y, 'f');  // 值：3.14
    print_value(&z, 'c');  // 值：A
    
    return 0;
}
```

1. **指针交换数组元素**：
   ```c
   #include <stdio.h>
   void reverse_array(int *arr, int size) {
       int *left = arr;
       int *right = arr + size - 1;
       while (left < right) {
           int temp = *left;
           *left = *right;
           *right = temp;
           left++;
           right--;
       }
   }
   int main(void) {
       int arr[] = {1, 2, 3, 4, 5};
       int size = sizeof(arr) / sizeof(arr[0]);
       reverse_array(arr, size);
       for (int i = 0; i < size; i++) {
           printf("%d ", arr[i]);
       }
       printf("\n");
       return 0;
   }
   ```

2. **字符串数组排序**：
   ```c
   #include <stdio.h>
   #include <string.h>
   void sort_strings(const char *arr[], int size) {
       for (int i = 0; i < size - 1; i++) {
           for (int j = 0; j < size - 1 - i; j++) {
               if (strcmp(arr[j], arr[j + 1]) > 0) {
                   const char *temp = arr[j];
                   arr[j] = arr[j + 1];
                   arr[j + 1] = temp;
               }
           }
       }
   }
   int main(void) {
       const char *names[] = {"Charlie", "Alice", "Bob"};
       int size = sizeof(names) / sizeof(names[0]);
       sort_strings(names, size);
       for (int i = 0; i < size; i++) {
           printf("%s\n", names[i]);
       }
       return 0;
   }
   ```

3. **函数指针实现计算器**：
   ```c
   #include <stdio.h>
   double add(double a, double b) { return a + b; }
   double sub(double a, double b) { return a - b; }
   double mul(double a, double b) { return a * b; }
   double div(double a, double b) { return b != 0 ? a / b : 0; }
   
   int main(void) {
       double (*ops[4])(double, double) = {add, sub, mul, div};
       char symbols[] = {'+', '-', '*', '/'};
       
       for (int i = 0; i < 4; i++) {
           printf("10 %c 5 = %.2f\n", symbols[i], ops[i](10, 5));
       }
       return 0;
   }
   ```

## 本讲小结

| 概念 | 要点 |
|------|------|
| 指针 | 存储内存地址的变量 |
| & 取地址 | 获取变量的地址 |
| * 解引用 | 通过地址访问值 |
| 指针算术 | `p + n` 跳过 n 个元素 |

| 函数指针 | 存储函数地址，用于回调 |
| void* | 通用指针，可指向任何类型 |
