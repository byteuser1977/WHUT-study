## 1. 函数基础

函数是**完成特定任务的代码块**，就像菜谱中的一个步骤。

### 1.1 函数定义

```c
// 格式：
// 返回类型 函数名(参数列表) {
//     函数体
//     return 返回值;
// }

// 无参数无返回值
void greet(void) {
    printf("Hello!\n");
}

// 有参数有返回值
int add(int a, int b) {
    return a + b;
}

// 多参数
double calculate(double radius) {
    return 3.14159 * radius * radius;
}
```

### 1.2 函数声明（原型）

```c
// 告诉编译器函数的接口，放在main前面
int add(int a, b);  // 声明（原型）
// 或
int add(int, int);  // 参数名可省略

int main(void) {
    int result = add(3, 5);  // 调用
    printf("3 + 5 = %d\n", result);
    return 0;
}

// 函数定义（实现）
int add(int a, int b) {
    return a + b;
}
```

## 2. 参数传递

### 2.1 值传递

```c
// C语言只有值传递！
void swap(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
    printf("swap中：a=%d, b=%d\n", a, b);
}

int main(void) {
    int x = 10, y = 20;
    swap(x, y);
    printf("swap后：x=%d, y=%d\n", x, y);  // x和y没有交换！
    return 0;
}
// 输出：
// swap中：a=20, b=10
// swap后：x=10, y=20
```

### 2.2 数组传参

```c
// 数组传参时退化为指针，可以修改原数组
void modifyArray(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        arr[i] *= 2;  // 修改会影响原数组
    }
}

int main(void) {
    int nums[] = {1, 2, 3, 4, 5};
    int size = sizeof(nums) / sizeof(nums[0]);
    modifyArray(nums, size);
    for (int i = 0; i < size; i++) {
        printf("%d ", nums[i]);  // 输出：2 4 6 8 10
    }
    return 0;
}
```

## 3. 递归函数

函数可以调用自身，这叫递归。

### 3.1 递归三要素

1. **终止条件**：什么时候不再递归
2. **递归关系**：如何缩小问题规模
3. **递归调用**：调用自身

### 3.2 经典递归示例

```c
// 阶乘：n! = n × (n-1)!
int factorial(int n) {
    if (n <= 1) return 1;  // 终止条件
    return n * factorial(n - 1);  // 递归关系
}

// 斐波那契数列：f(n) = f(n-1) + f(n-2)
int fibonacci(int n) {
    if (n <= 1) return n;  // 终止条件
    return fibonacci(n - 1) + fibonacci(n - 2);  // 递归关系
}

// 汉诺塔
void hanoi(int n, char from, char to, char aux) {
    if (n == 1) {
        printf("移动盘子1从%c到%c\n", from, to);
        return;
    }
    hanoi(n - 1, from, aux, to);
    printf("移动盘子%d从%c到%c\n", n, from, to);
    hanoi(n - 1, aux, to, from);
}
```

## 4. 变量作用域和生存期

### 4.1 作用域

```c
int global = 10;  // 全局变量：整个文件可见

int main(void) {
    int local = 20;  // 局部变量：只在main内可见
    
    if (1) {
        int block = 30;  // 块作用域：只在if块内可见
        printf("%d %d %d\n", global, local, block);
    }
    // printf("%d\n", block);  // ❌ 错误：block不可见
    
    return 0;
}
```

### 4.2 生存期

```c
int global = 10;  // 静态存储期：程序运行期间一直存在

void func(void) {
    static int count = 0;  // 静态局部变量：只初始化一次，程序结束时销毁
    count++;
    printf("调用次数：%d\n", count);
}

int main(void) {
    int local = 20;  // 自动存储期：进入作用域时创建，离开时销毁
    
    func();  // 调用次数：1
    func();  // 调用次数：2
    func();  // 调用次数：3
    
    return 0;
}
```

## 5. 存储类别

| 存储类别 | 关键字 | 存储期 | 作用域 | 链接 | 初始值 |
|---------|--------|--------|--------|------|--------|
| 自动变量 | `auto`（可省略） | 自动 | 块 | 无 | 未定义 |
| 寄存器变量 | `register` | 自动 | 块 | 无 | 未定义 |
| 静态局部变量 | `static` | 静态 | 块 | 无 | 0 |
| 外部变量 | `extern` | 静态 | 文件 | 外部 | 0 |

### 5.1 static 关键字

```c
// 静态局部变量：只初始化一次，函数调用间保持值
void counter(void) {
    static int count = 0;  // 只执行一次
    count++;
    printf("count = %d\n", count);
}

// 静态全局变量：限制在本文件可见
static int file_private = 42;  // 其他文件不能用extern访问
```

### 5.2 extern 关键字

```c
// 声明在其他文件定义的变量
extern int global_var;  // 在其他文件中定义
```

1. **递归求和**：
   ```c
   #include <stdio.h>
   int sum(int n) {
       if (n <= 1) return n;
       return n + sum(n - 1);
   }
   int main(void) {
       printf("1+2+...+10 = %d\n", sum(10));
       return 0;
   }
   ```

2. **递归求幂**：
   ```c
   #include <stdio.h>
   int power(int base, int exp) {
       if (exp == 0) return 1;
       return base * power(base, exp - 1);
   }
   int main(void) {
       printf("2^10 = %d\n", power(2, 10));
       return 0;
   }
   ```

3. **static变量应用**：
   ```c
   #include <stdio.h>
   int unique_id(void) {
       static int id = 0;
       return ++id;
   }
   int main(void) {
       printf("ID1: %d\n", unique_id());
       printf("ID2: %d\n", unique_id());
       printf("ID3: %d\n", unique_id());
       return 0;
   }
   ```
